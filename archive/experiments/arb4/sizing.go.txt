package main

import (
	pf "defi-toolbox/pathfinder"
	"defi-toolbox/statedb"
	"math/big"
	"math"

	"github.com/ava-labs/libevm/common"
	"github.com/holiman/uint256"
)

const sizingRoundSamples = 9

// EVMSizing runs a two-pass logarithmic grid search on the winning cycle's input amount
// using pure EVM calls. Each evaluation is one full-path swap() — exact output and gas.
//
// Round 1 samples the full range on a log scale.
// Round 2 resamples the local bracket around the best round-1 point.
func EVMSizing(
	steps []pf.RouteStep,
	stateWithOverrides *statedb.StateDB,
	cfg statedb.EVMConfig,
	routerAddr, sender common.Address,
	hub common.Address,
	gasPrice uint64,
	prices map[common.Address]float64,
	seedAmountIn *uint256.Int,
	maxAmountIn *uint256.Int,
) *CycleResult {
	evalEVM := makeEVMEvaluator(steps, stateWithOverrides, cfg, routerAddr, sender, hub, gasPrice, prices)

	// Search range based on hub token
	lo := toWei(1, 15)    // 0.001 AVAX
	hi := toWei(1000, 18) // 1000 AVAX
	switch hub {
	case USDC, USDT:
		lo = toWei(1, 4)      // 0.01 USDC
		hi = toWei(100000, 6) // 100k USDC
	case WETHe:
		lo = toWei(1, 13)   // 0.00001 ETH
		hi = toWei(100, 18) // 100 ETH
	}
	if maxAmountIn != nil && maxAmountIn.Cmp(hi) < 0 {
		hi = new(uint256.Int).Set(maxAmountIn)
	}
	if hi.Cmp(lo) < 0 || hi.IsZero() {
		return nil
	}

	var best *CycleResult
	if seedAmountIn != nil && seedAmountIn.Cmp(lo) >= 0 && seedAmountIn.Cmp(hi) <= 0 {
		best = evalEVM(seedAmountIn)
	}

	round1 := logSamples(lo, hi, sizingRoundSamples)
	bestIdx := -1
	for i, amt := range round1 {
		res := evalEVM(amt)
		if res != nil && (best == nil || res.Profit.Cmp(&best.Profit) > 0) {
			best = res
			bestIdx = i
		}
	}
	if best == nil {
		return nil
	}

	refineLo := new(uint256.Int).Set(round1[maxInt(bestIdx-1, 0)])
	refineHi := new(uint256.Int).Set(round1[minInt(bestIdx+1, len(round1)-1)])
	if refineLo.Cmp(refineHi) == 0 {
		return best
	}

	for _, amt := range logSamples(refineLo, refineHi, sizingRoundSamples) {
		res := evalEVM(amt)
		if res != nil && (best == nil || res.Profit.Cmp(&best.Profit) > 0) {
			best = res
		}
	}
	return best
}

func makeEVMEvaluator(
	steps []pf.RouteStep,
	stateWithOverrides *statedb.StateDB,
	cfg statedb.EVMConfig,
	routerAddr, sender common.Address,
	hub common.Address,
	gasPrice uint64,
	prices map[common.Address]float64,
) func(amountIn *uint256.Int) *CycleResult {
	evmCtx := statedb.GetCachedContext(cfg)
	return func(amountIn *uint256.Int) *CycleResult {
		calldata := buildCalldata(steps, amountIn)
		cs := statedb.NewCallState(stateWithOverrides)
		ret, gasUsed, err := evmCtx.ExecuteWithCallState(cs, sender, routerAddr, calldata)
		if err != nil || len(ret) < 32 {
			return nil
		}
		var evmOut uint256.Int
		evmOut.SetBytes(ret[:32])
		if evmOut.Bytes32()[0]&0x80 != 0 || evmOut.IsZero() {
			return nil
		}
		gasCostHub, ok := GasCostInTokenBig(gasUsed, gasPrice, hub, prices)
		if !ok {
			return nil
		}
		profitBig := new(big.Int).Set(evmOut.ToBig())
		profitBig.Sub(profitBig, amountIn.ToBig())
		profitBig.Sub(profitBig, gasCostHub)
		if profitBig.Sign() <= 0 {
			return nil
		}
		profit := new(uint256.Int)
		profit.SetFromBig(profitBig)
		return &CycleResult{
			Hub:       hub,
			Steps:     steps,
			AmountIn:  *amountIn,
			AmountOut: evmOut,
			Profit:    *profit,
			GasUsed:   gasUsed,
			Calldata:  calldata,
		}
	}
}

func logSamples(lo, hi *uint256.Int, count int) []*uint256.Int {
	if count <= 1 || lo.Cmp(hi) >= 0 {
		return []*uint256.Int{new(uint256.Int).Set(lo), new(uint256.Int).Set(hi)}
	}

	loF := u256ToFloat(lo)
	hiF := u256ToFloat(hi)
	if !(loF > 0) || !(hiF > loF) {
		return []*uint256.Int{new(uint256.Int).Set(lo), new(uint256.Int).Set(hi)}
	}

	logLo := math.Log(loF)
	logHi := math.Log(hiF)
	seen := make(map[string]bool, count)
	samples := make([]*uint256.Int, 0, count)
	for i := 0; i < count; i++ {
		t := float64(i) / float64(count-1)
		value := math.Exp(logLo + (logHi-logLo)*t)
		amt := floatToU256(value)
		if amt.Cmp(lo) < 0 {
			amt = new(uint256.Int).Set(lo)
		}
		if amt.Cmp(hi) > 0 {
			amt = new(uint256.Int).Set(hi)
		}
		key := amt.Dec()
		if !seen[key] {
			seen[key] = true
			samples = append(samples, amt)
		}
	}
	if len(samples) == 0 {
		return []*uint256.Int{new(uint256.Int).Set(lo), new(uint256.Int).Set(hi)}
	}
	if samples[0].Cmp(lo) != 0 {
		samples = append([]*uint256.Int{new(uint256.Int).Set(lo)}, samples...)
	}
	if samples[len(samples)-1].Cmp(hi) != 0 {
		samples = append(samples, new(uint256.Int).Set(hi))
	}
	return samples
}

func u256ToFloat(v *uint256.Int) float64 {
	if v == nil || v.IsZero() {
		return 0
	}
	f, _ := new(big.Float).SetInt(v.ToBig()).Float64()
	return f
}

func floatToU256(v float64) *uint256.Int {
	if !(v > 0) {
		return uint256.NewInt(0)
	}
	bf := new(big.Float).SetPrec(256).SetFloat64(v)
	bi, _ := bf.Int(nil)
	if bi == nil || bi.Sign() <= 0 {
		return uint256.NewInt(0)
	}
	out := new(uint256.Int)
	out.SetFromBig(bi)
	return out
}

func minInt(a, b int) int {
	if a < b {
		return a
	}
	return b
}

func maxInt(a, b int) int {
	if a > b {
		return a
	}
	return b
}
